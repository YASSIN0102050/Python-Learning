1. Lists – Advanced Operations

Lists are mutable (تتغير). يمكنك إضافة/حذف/تعديل العناصر بسهولة. 
tasks = ["Study", "Code", "Sleep"]

tasks.append("Eat")           # add at end
tasks.insert(1, "Exercise")   # add at index 1
tasks.remove("Sleep")         # remove by value
popped = tasks.pop()          # remove & return last item

print(tasks)                  # ['Study', 'Exercise', 'Code', 'Eat']
print("Length:", len(tasks))
print("Sorted:", sorted(tasks))

2. List Comprehensions (أسرع وأنظف طريقة)

# Normal way
squares = []
for i in range(10):
    squares.append(i**2)

# List comprehension (الطريقة المفضلة)
squares = [i**2 for i in range(10)]

# with condition
even_squares = [i**2 for i in range(20) if i % 2 == 0]

print(even_squares)

3. Dictionaries – Key → Value

مثل قاموس: كل كلمة (key) لها معنى (value). 
student = {
    "name": "Yassin",
    "age": 19,
    "grades": [85, 92, 78],
    "is_active": True
}

print(student["name"])
student["city"] = "Cairo"          # add new key
student["age"] += 1                # update
print(student.get("phone", "Not found"))  # safe get

for key, value in student.items():
    print(f"{key}: {value}")

4. Tuples & Sets

# Tuple → لا يتغير (immutable) وأسرع من list
coords = (30.0444, 31.2357)   # latitude, longitude
x, y = coords                 # unpacking

# Set → لا تكرر + عمليات رياضية سريعة
a = {1, 2, 3, 3}             # → {1,2,3}
b = {3, 4, 5}
print(a | b)                 # union  {1,2,3,4,5}
print(a & b)                 # intersection {3}
print(a - b)                 # difference {1,2}

5. Functions – Advanced

def calculate_total(items, discount=0.1):
    total = sum(items)
    return total * (1 - discount)

print(calculate_total([100, 200, 50]))           # 315.0
print(calculate_total([300, 400], 0.2))          # 560.0

# *args → عدد متغير من الـ arguments
def sum_all(*numbers):
    return sum(numbers)

print(sum_all(1,2,3,4,5))   # 15

6. File Handling + with statement

# Write
with open("notes.txt", "w") as file:
    file.write("Python is awesome!\nLine 2\n")

# Read
with open("notes.txt", "r") as file:
    content = file.read()
    print(content)

# Read line by line
with open("notes.txt") as f:
    for line in f:
        print(line.strip())

7. Mini Project – To-Do List (save/load from file)

def load_tasks():
    try:
        with open("tasks.txt", "r") as f:
            return [line.strip() for line in f if line.strip()]
    except FileNotFoundError:
        return []

def save_tasks(tasks):
    with open("tasks.txt", "w") as f:
        for task in tasks:
            f.write(task + "\n")

tasks = load_tasks()

while True:
    print("\n1. Show tasks\n2. Add task\n3. Remove task\n4. Exit")
    choice = input("Choose: ")
    
    if choice == "1":
        for i, t in enumerate(tasks, 1):
            print(f"{i}. {t}")
    elif choice == "2":
        task = input("New task: ")
        tasks.append(task)
        save_tasks(tasks)
    elif choice == "3":
        num = int(input("Task number to remove: ")) - 1
        if 0 <= num < len(tasks):
            tasks.pop(num)
            save_tasks(tasks)
    elif choice == "4":
        break
